   277→function AddExpenseForm({ onDone }: { onDone: () => void }) {
   278→  const { data: trip } = useTrip();
   279→  const add = useAddExpense();
   280→  const [amount, setAmount] = useState("");
   281→  const [category, setCategory] = useState("food");
   282→  const [description, setDescription] = useState("");
   283→  const [paidById, setPaidById] = useState(trip?.settings.currentUserId ?? "");
   284→  const [dayId, setDayId] = useState("");
   285→
   286→  const submit = async () => {
   287→    const amt = parseFloat(amount);
   288→    if (!amt || !description) {
   289→      toast.error("Заполните сумму и описание");
   290→      return;
   291→    }
   292→    await add.mutateAsync({
   293→      amount: amt,
   294→      category,
   295→      description,
   296→      paidById,
   297→      dayId: dayId || undefined,
   298→    });
   299→    toast.success("Трата добавлена 💸");
   300→    setAmount(""); setDescription("");
   301→    onDone();
   302→  };
   303→
   304→  return (
   305→    <motion.div
   306→      initial={{ height: 0, opacity: 0 }}
   307→      animate={{ height: "auto", opacity: 1 }}
   308→      exit={{ height: 0, opacity: 0 }}
   309→      className="overflow-hidden"
   310→    >
   311→      <div className="bg-muted/40 rounded-xl p-3 space-y-2 mb-2">
   312→        <div className="grid grid-cols-2 gap-2">
   313→          <input
   314→            type="number"
   315→            value={amount}
   316→            onChange={(e) => setAmount(e.target.value)}
   317→            placeholder="Сумма $"
   318→            className="rounded-lg border border-input bg-background px-3 py-2 text-sm"

   358→function ParticipantBudgetRow({ participant, spent }: { participant: Participant; spent: number }) {
   359→  const update = useUpdateMember();
   360→  const tripId = getTripId();
   361→  const [editing, setEditing] = useState(false);
   362→  const [val, setVal] = useState(participant.budget?.toString() ?? "");
   363→
   364→  const budget = participant.budget;
   365→  const remaining = budget !== null ? budget - spent : null;
   366→  const pct = budget && budget > 0 ? Math.min(100, (spent / budget) * 100) : null;
   367→
   368→  const save = () => {
   369→    const num = val.trim() ? parseFloat(val) : null;
   370→    // participant.id — это userId, нужно найти memberId из trip.members
   371→    // Но в текущей архитектуре participants уже имеют id = userId
   372→    // Используем tripId + participant.id как memberId (упрощение)
   373→    update.mutate({ memberId: participant.id, tripId, budget: num });
   374→    toast.success("Бюджет обновлён");
   375→    setEditing(false);
   376→  };
   377→
   378→  return (
   379→    <div className="flex items-center gap-2.5 p-2 rounded-lg hover:bg-accent/50">
   380→      <div className="size-8 rounded-full grid place-items-center text-sm shrink-0" style={{ background: participant.color }}>
   381→        {participant.emoji}
   382→      </div>
   383→      <div className="flex-1 min-w-0">
   384→        <div className="text-sm font-medium">{participant.name}</div>
   385→        <div className="text-[11px] text-muted-foreground">потратил ${spent.toFixed(0)}</div>
   386→        {pct !== null && (
   387→          <div className="mt-1 h-1 rounded-full bg-muted overflow-hidden max-w-[120px]">
   388→            <div
   389→              className={cn("h-full rounded-full", pct > 90 ? "bg-red-500" : pct > 70 ? "bg-amber-500" : "bg-green-500")}
   390→              style={{ width: `${pct}%` }}
   391→            />
   392→          </div>
   393→        )}
   394→      </div>
   395→      {editing ? (
   396→        <div className="flex items-center gap-1">
   397→          <input
   398→            type="number"
   399→            inputMode="decimal"
   400→            value={val}
   401→            onChange={(e) => setVal(e.target.value)}
   402→            onKeyDown={(e) => { if (e.key === "Enter") save(); if (e.key === "Escape") setEditing(false); }}
   403→            onBlur={save}
   404→            autoFocus
   405→            placeholder="—"
   406→            className="w-20 text-sm rounded-lg border border-input bg-background px-2 py-1 text-right"
   407→          />
   408→        </div>
   409→      ) : (
   410→        <button
   411→          onClick={() => { setVal(participant.budget?.toString() ?? ""); setEditing(true); }}
   412→          className="text-right group"
   413→        >
   414→          <div className={cn("text-sm font-semibold", remaining !== null && remaining < 0 && "text-red-500")}>
   415→            {budget !== null ? `$${budget}` : "—"}
   416→          </div>
   417→          {remaining !== null && (
   418→            <div className={cn("text-[10px]", remaining < 0 ? "text-red-500" : "text-muted-foreground")}>
   419→              ост. ${remaining.toFixed(0)}
   420→            </div>
   421→          )}
   422→          <Pencil className="size-2.5 text-muted-foreground/0 group-hover:text-muted-foreground transition-colors inline-block ml-1" />
   423→        </button>
   424→      )}
   425→    </div>
   426→  );
   427→}

   429→// Алгоритм упрощения долгов (greedy)
   430→function BudgetHero({ totalSpent, totalBudget, budgetPct, remaining }: {
   431→  totalSpent: number;
   432→  totalBudget: number;
   433→  budgetPct: number;
   434→  remaining: number;
   435→}) {
   436→  const update = useUpdateTripBudget();
   437→  const [editing, setEditing] = useState(false);
   438→  const [val, setVal] = useState(String(totalBudget));
   439→
   440→  const save = () => {
   441→    const num = parseFloat(val);
   442→    if (!isNaN(num) && num >= 0) {
   443→      update.mutate(num);
   444→      toast.success("Бюджет обновлён");
   445→    }
   446→    setEditing(false);

   510→          </span>

   517→function settleDebts(balances: { participant: Participant; paid: number; balance: number }[]) {
   518→  const creditors = balances.filter((b) => b.balance > 0.01).sort((a, b) => b.balance - a.balance);
   519→  const debtors = balances.filter((b) => b.balance < -0.01).sort((a, b) => a.balance - b.balance);
   520→  const result: { from: Participant; to: Participant; amount: number }[] = [];
   521→  let i = 0, j = 0;
   522→  const c = creditors.map((x) => ({ ...x }));
   523→  const d = debtors.map((x) => ({ ...x }));
   524→  while (i < d.length && j < c.length) {
   525→    const amt = Math.min(-d[i].balance, c[j].balance);
   526→    result.push({ from: d[i].participant, to: c[j].participant, amount: amt });
   527→    d[i].balance += amt;
   528→    c[j].balance -= amt;
   529→    if (Math.abs(d[i].balance) < 0.01) i++;
   530→    if (Math.abs(c[j].balance) < 0.01) j++;
   531→  }
   532→  return result;
   533→}
