   857→function ParticipantBudgetRow({ participant, spent }: { participant: Participant; spent: number }) {

  1017→function MarkSettledButton({ from, to, amount }: { from: Participant; to: Participant; amount: number }) {

  1025→  const isDebtor = currentUserId === from.id; // Я должен → жду подтверждения

  1051→  // Кредитор (ему должны) — может нажать "Перевели" чтобы подтвердить получение
  1052→  if (isCreditor) {

  1056→        disabled={addExpense.isPending}
  1057→        className="shrink-0 text-[10px] bg-green-600/10 text-green-600 hover:bg-green-600/20 px-2.5 py-1.5 rounded-lg font-medium flex items-center gap-1 transition-colors active:scale-95"

  1070→        <Clock className="size-3" /> Ждём подтверждения
  1071→      </span>
  1072→    );
  1073→  }
  1074→
  1075→  // Если я ни при чём — ничего не показываем

  1079→function settleDebts(balances: { participant: Participant; paid: number; balance: number }[]) {
  1080→  const creditors = balances.filter((b) => b.balance > 0.01).sort((a, b) => b.balance - a.balance);
  1081→  const debtors = balances.filter((b) => b.balance < -0.01).sort((a, b) => a.balance - b.balance);
  1082→  const result: { from: Participant; to: Participant; amount: number }[] = [];
  1083→  let i = 0, j = 0;
  1084→  const c = creditors.map((x) => ({ ...x }));
  1085→  const d = debtors.map((x) => ({ ...x }));
  1086→  while (i < d.length && j < c.length) {
  1087→    const amt = Math.min(-d[i].balance, c[j].balance);
  1088→    result.push({ from: d[i].participant, to: c[j].participant, amount: amt });
  1089→    d[i].balance += amt;
  1090→    c[j].balance -= amt;
  1091→    if (Math.abs(d[i].balance) < 0.01) i++;
  1092→    if (Math.abs(c[j].balance) < 0.01) j++;
  1093→  }
  1094→  return result;
  1095→}
