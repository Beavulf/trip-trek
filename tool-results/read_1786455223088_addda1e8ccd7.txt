  1079→  1079→function settleDebts(balances: { participant: Participant; paid: number; balance: number }[]) {
  1080→  1080→  const creditors = balances.filter((b) => b.balance > 0.01).sort((a, b) => b.balance - a.balance);
  1081→  1081→  const debtors = balances.filter((b) => b.balance < -0.01).sort((a, b) => a.balance - b.balance);
  1082→  1082→  const result: { from: Participant; to: Participant; amount: number }[] = [];
  1083→  1083→  let i = 0, j = 0;
  1084→  1084→  const c = creditors.map((x) => ({ ...x }));
  1085→  1085→  const d = debtors.map((x) => ({ ...x }));
  1086→  1086→  while (i < d.length && j < c.length) {
  1087→  1087→    const amt = Math.min(-d[i].balance, c[j].balance);
  1088→  1088→    result.push({ from: d[i].participant, to: c[j].participant, amount: amt });
  1089→  1089→    d[i].balance += amt;
  1090→  1090→    c[j].balance -= amt;
  1091→  1091→    if (Math.abs(d[i].balance) < 0.01) i++;
  1092→  1092→    if (Math.abs(c[j].balance) < 0.01) j++;
  1093→  1093→  }
  1094→  1094→  return result;
  1095→  1095→}
