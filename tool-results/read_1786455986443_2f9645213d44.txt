   819→// Push toggle component
   820→function PushToggle() {
   821→  const [enabled, setEnabled] = useState(false);
   822→  const [loading, setLoading] = useState(false);
   823→  const [supported] = useState(() => {
   824→    if (typeof window === "undefined") return false;
   825→    return "Notification" in window && "serviceWorker" in navigator && "PushManager" in window;
   826→  });
   827→  const { data: session } = useSession();
   828→  const userId = (session?.user as { id?: string } | undefined)?.id || "";
   829→
   830→  useEffect(() => {
   831→    if (!supported) return;
   832→    checkStatus();
   833→  }, [supported]);
   834→
   835→  const checkStatus = async () => {
   836→    try {
   837→      // Permission должен быть granted
   838→      if (Notification.permission !== "granted") {
   839→        setEnabled(false);
   840→        return;
   841→      }
   842→      // Проверяем подписку
   843→      const reg = await navigator.serviceWorker.ready;
   844→      const sub = await reg.pushManager.getSubscription();
   845→      setEnabled(!!sub);
   846→    } catch {
   847→      setEnabled(false);
   848→    }
   849→  };
   850→
   851→  const toggle = async () => {
   852→    if (!supported) {
   853→      toast.error("Push не поддерживается", {
   854→        description: "Открой через Chrome/Safari, не через встроенный браузер",
   855→      });
   856→      return;
   857→    }
   858→
   859→    setLoading(true);
   860→    try {
   861→      if (enabled) {
   862→        // ВЫКЛЮЧАЕМ — удаляем все подписки
   863→        try {
   864→          const reg = await navigator.serviceWorker.ready;
   865→          const sub = await reg.pushManager.getSubscription();
   866→          if (sub) {
   867→            await sub.unsubscribe();
   868→            await fetch(`/api/push/subscribe?endpoint=${encodeURIComponent(sub.endpoint)}`, {
   869→              method: "DELETE",
   870→            });
   871→          }
   872→        } catch (e) {
   873→          console.warn("SW unsubscribe error:", e);
   874→        }
   875→        // Также удаляем все подписки пользователя с сервера
   876→        await fetch(`/api/push/subscribe?userId=${userId}`, { method: "DELETE" }).catch(() => {});
   877→        setEnabled(false);
   878→        toast.success("Уведомления отключены");
   879→      } else {
   880→        // ВКЛЮЧАЕМ
   881→        // Сначала регистрируем SW если ещё не зарегистрирован
   882→        await navigator.serviceWorker.register("/sw.js");
   883→        await navigator.serviceWorker.ready;
   884→
   885→        // Запрашиваем permission
   886→        const perm = await Notification.requestPermission();
   887→        if (perm !== "granted") {
   888→          toast.error("Разрешение не дано", {
   889→            description: "Разреши уведомления в настройках браузера",
   890→          });
   891→          return;
   892→        }
   893→
   894→        // Получаем VAPID ключ
   895→        const vapidRes = await fetch("/api/push/vapid-public-key");
   896→        const { publicKey } = await vapidRes.json();
   897→        if (!publicKey) {
   898→          toast.error("Push не настроен на сервере");
   899→          return;
   900→        }
   901→
   902→        // Подписываемся
   903→        const reg = await navigator.serviceWorker.ready;
   904→        const subscription = await reg.pushManager.subscribe({
   905→          userVisibleOnly: true,
   906→          applicationServerKey: urlBase64ToUint8Array(publicKey) as BufferSource,
   907→        });
   908→
   909→        await fetch("/api/push/subscribe", {
   910→          method: "POST",
   911→          headers: { "Content-Type": "application/json" },
   912→          body: JSON.stringify({
   913→            userId,
   914→            subscription: subscription.toJSON(),
   915→          }),
   916→        });
   917→
   918→        setEnabled(true);
   919→        toast.success("Уведомления включены 🔔", {
   920→          description: "Будешь получать push даже при закрытом приложении",
   921→        });
   922→      }
   923→    } catch (e) {
   924→      console.error("Push error:", e);
   925→      toast.error("Ошибка", {
   926→        description: "Попробуй перезагрузить страницу и снова",
   927→      });
   928→    } finally {
   929→      setLoading(false);
   930→    }
   931→  };
   932→
   933→  return (
   934→    <button
   935→      onClick={toggle}
   936→      disabled={loading}
   937→      className={cn(
   938→        "w-10 h-6 rounded-full relative transition-colors shrink-0",
   939→        enabled ? "bg-primary" : "bg-muted"
   940→      )}
   941→    >
   942→      <div className={cn(
   943→        "absolute top-0.5 size-5 rounded-full bg-white shadow transition-transform",
   944→        enabled ? "translate-x-4" : "translate-x-0.5"
   945→      )} />
   946→    </button>
   947→  );
   948→}

   950→function urlBase64ToUint8Array(base64String: string): Uint8Array {
   951→  const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
   952→  const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
   953→  const rawData = atob(base64);
   954→  const outputArray = new Uint8Array(rawData.length);
   955→  for (let i = 0; i < rawData.length; ++i) {
   956→    outputArray[i] = rawData.charCodeAt(i);
   957→  }
   958→  return outputArray;
   959→}
