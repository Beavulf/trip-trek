   825→   825→    return "Notification" in window && "serviceWorker" in navigator && "PushManager" in window;
   826→   826→  });
   827→   827→  const { data: session } = useSession();
   828→   828→  const userId = (session?.user as { id?: string } | undefined)?.id || "";
   829→   829→
   830→   830→  useEffect(() => {
   831→   831→    if (!supported) return;
   832→   832→    checkStatus();
   833→   833→  }, [supported]);
   834→   834→
   835→   835→  const checkStatus = async () => {
   836→   836→    try {
   837→   837→      // Permission должен быть granted
   838→   838→      if (Notification.permission !== "granted") {
   839→   839→        setEnabled(false);
   840→   840→        return;
   841→   841→      }
   842→   842→      // Проверяем подписку
   843→   843→      const reg = await navigator.serviceWorker.ready;
   844→   844→      const sub = await reg.pushManager.getSubscription();
   845→   845→      setEnabled(!!sub);
   846→   846→    } catch {
   847→   847→      setEnabled(false);
   848→   848→    }
   849→   849→  };
   850→   850→
   851→   851→  const toggle = async () => {
   852→   852→    if (!supported) {
   853→   853→      toast.error("Push не поддерживается", {
   854→   854→        description: "Открой через Chrome/Safari, не через встроенный браузер",
   855→   855→      });
   856→   856→      return;
   857→   857→    }
   858→   858→
   859→   859→    setLoading(true);
   860→   860→    try {
   861→   861→      if (enabled) {
   862→   862→        // ВЫКЛЮЧАЕМ — удаляем все подписки
   863→   863→        try {
   864→   864→          const reg = await navigator.serviceWorker.ready;
   865→   865→          const sub = await reg.pushManager.getSubscription();
   866→   866→          if (sub) {
   867→   867→            await sub.unsubscribe();
   868→   868→            await fetch(`/api/push/subscribe?endpoint=${encodeURIComponent(sub.endpoint)}`, {
   869→   869→              method: "DELETE",
   870→   870→            });
   871→   871→          }
   872→   872→        } catch (e) {
   873→   873→          console.warn("SW unsubscribe error:", e);
   874→   874→        }
   875→   875→        // Также удаляем все подписки пользователя с сервера
   876→   876→        await fetch(`/api/push/subscribe?userId=${userId}`, { method: "DELETE" }).catch(() => {});
   877→   877→        setEnabled(false);
   878→   878→        toast.success("Уведомления отключены");
   879→   879→      } else {
   880→   880→        // ВКЛЮЧАЕМ
   881→   881→        // Сначала регистрируем SW если ещё не зарегистрирован
   882→   882→        await navigator.serviceWorker.register("/sw.js");
   883→   883→        await navigator.serviceWorker.ready;
   884→   884→
   885→   885→        // Запрашиваем permission
   886→   886→        const perm = await Notification.requestPermission();
   887→   887→        if (perm !== "granted") {
   888→   888→          toast.error("Разрешение не дано", {
   889→   889→            description: "Разреши уведомления в настройках браузера",
   890→   890→          });
   891→   891→          return;
   892→   892→        }
   893→   893→
   894→   894→        // Получаем VAPID ключ
   895→   895→        const vapidRes = await fetch("/api/push/vapid-public-key");
   896→   896→        const { publicKey } = await vapidRes.json();
   897→   897→        if (!publicKey) {
   898→   898→          toast.error("Push не настроен на сервере");
   899→   899→          return;
   900→   900→        }
   901→   901→
   902→   902→        // Подписываемся
   903→   903→        const reg = await navigator.serviceWorker.ready;
   904→   904→        const subscription = await reg.pushManager.subscribe({
   905→   905→          userVisibleOnly: true,
   906→   906→          applicationServerKey: urlBase64ToUint8Array(publicKey) as BufferSource,
   907→   907→        });
   908→   908→
   909→   909→        await fetch("/api/push/subscribe", {
   910→   910→          method: "POST",
   911→   911→          headers: { "Content-Type": "application/json" },
   912→   912→          body: JSON.stringify({
   913→   913→            userId,
   914→   914→            subscription: subscription.toJSON(),
   915→   915→          }),
   916→   916→        });
   917→   917→
   918→   918→        setEnabled(true);
   919→   919→        toast.success("Уведомления включены 🔔", {
   920→   920→          description: "Будешь получать push даже при закрытом приложении",
   921→   921→        });
   922→   922→      }
   923→   923→    } catch (e) {
   924→   924→      console.error("Push error:", e);
   925→   925→      toast.error("Ошибка", {
   926→   926→        description: "Попробуй перезагрузить страницу и снова",
   927→   927→      });
   928→   928→    } finally {
   929→   929→      setLoading(false);
   930→   930→    }
   931→   931→  };
   932→   932→
   933→   933→  return (
   934→   934→    <button
   935→   935→      onClick={toggle}
   936→   936→      disabled={loading}
   937→   937→      className={cn(
   938→   938→        "w-10 h-6 rounded-full relative transition-colors shrink-0",
   939→   939→        enabled ? "bg-primary" : "bg-muted"
   940→   940→      )}
   941→   941→    >
   942→   942→      <div className={cn(
   943→   943→        "absolute top-0.5 size-5 rounded-full bg-white shadow transition-transform",
   944→   944→        enabled ? "translate-x-4" : "translate-x-0.5"
   945→   945→      )} />
   946→   946→    </button>
   947→   947→  );
   948→   948→}

   950→   950→function urlBase64ToUint8Array(base64String: string): Uint8Array {
   951→   951→  const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
   952→   952→  const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
   953→   953→  const rawData = atob(base64);
   954→   954→  const outputArray = new Uint8Array(rawData.length);
   955→   955→  for (let i = 0; i < rawData.length; ++i) {
   956→   956→    outputArray[i] = rawData.charCodeAt(i);
   957→   957→  }
   958→   958→  return outputArray;
   959→   959→}
