   377→function CountdownCard({
   378→  isBefore,
   379→  isAfter,
   380→  daysRemaining,
   381→  currentDay,
   382→  totalDays,
   383→  startDate,
   384→  endDate,
   385→}: {
   386→  isBefore: boolean;
   387→  isAfter: boolean;
   388→  daysRemaining: number;
   389→  currentDay: number;
   390→  totalDays: number;
   391→  startDate: string;
   392→  endDate: string | null;
   393→}) {
   394→  const [showEditor, setShowEditor] = useState(false);
   395→  let icon = <Clock className="size-5" />;
   396→  let label = "В пути";
   397→  let value = `${currentDay}/${totalDays}`;
   398→  let sub = `осталось ${daysRemaining} дн.`;
   399→  let color = "#10b981";
   400→
   401→  if (isBefore) {
   402→    icon = <Plane className="size-5" />;
   403→    label = "До поездки";
   404→    value = `${daysRemaining}`;
   405→    sub = daysRemaining === 1 ? "день" : daysRemaining < 5 ? "дня" : "дней";
   406→    color = "#f59e0b";
   407→  } else if (isAfter) {
   408→    icon = <Route className="size-5" />;
   409→    label = "Поездка завершена";
   410→    value = "✓";
   411→    sub = `${totalDays} дней позади`;
   412→    color = "#8b5cf6";
   413→  }
   414→
   415→  const startStr = new Date(startDate).toISOString().slice(0, 10);
   416→  const endStr = endDate ? new Date(endDate).toISOString().slice(0, 10) : "";
   417→
   418→  return (
   419→    <div className="col-span-2 lg:col-span-1 rounded-2xl bg-card border border-border p-4 relative overflow-hidden">
   420→      <div
   421→        className="absolute -top-3 -right-3 size-16 rounded-full opacity-10 blur-xl"
   422→        style={{ background: color }}
   423→      />
   424→      <button
   425→        onClick={() => setShowEditor((v) => !v)}
   426→        className="absolute top-2 right-2 size-6 rounded-md hover:bg-accent grid place-items-center text-muted-foreground z-10"
   427→        title="Изменить даты"
   428→      >
   429→        <Settings2 className="size-3.5" />
   430→      </button>
   431→      <div className="flex items-center gap-2 text-sm text-muted-foreground mb-2">
   432→        <span style={{ color }}>{icon}</span> {label}
   433→      </div>
   434→      <div className="flex items-baseline gap-1.5">
   435→        <span className="text-3xl font-bold leading-none" style={{ color }}>{value}</span>
   436→        <span className="text-xs text-muted-foreground">{sub}</span>
   437→      </div>
   438→      {/* мини-таймлайн */}
   439→      <div className="mt-3 flex gap-0.5">
   440→        {Array.from({ length: totalDays }).map((_, i) => (
   441→          <div
   442→            key={i}
   443→            className={cn(
   444→              "h-1.5 flex-1 rounded-full transition-colors",
   445→              isBefore ? "bg-muted" : isAfter ? "bg-primary" : i + 1 < currentDay ? "bg-primary" : i + 1 === currentDay ? "bg-primary/60" : "bg-muted"
   446→            )}
   447→          />
   448→        ))}
   449→      </div>
   450→
   451→      {showEditor && <DatesEditor startStr={startStr} endStr={endStr} onDone={() => setShowEditor(false)} />}
   452→    </div>
   453→  );
   454→}
   455→
   456→function DatesEditor({ startStr, endStr, onDone }: { startStr: string; endStr: string; onDone: () => void }) {
   457→  const update = useUpdateTripDates();
   458→  const [start, setStart] = useState(startStr);
   459→  const [end, setEnd] = useState(endStr);
   460→
   461→  const save = async () => {
   462→    await update.mutateAsync({ startDate: start, endDate: end || undefined });
   463→    toast.success("Даты обновлены");
   464→    onDone();
   465→  };
   466→
   467→  return (
   468→    <div className="mt-3 pt-3 border-t border-border space-y-2 relative z-20">
   469→      <div className="grid grid-cols-2 gap-2">
   470→        <div>
   471→          <label className="text-[10px] text-muted-foreground block mb-0.5">Вылет ✈️</label>
   472→          <input
   473→            type="date"
   474→            value={start}
   475→            onChange={(e) => setStart(e.target.value)}
   476→            className="w-full text-xs rounded-lg border border-input bg-background px-2 py-1.5"
   477→          />
   478→        </div>
   479→        <div>
   480→          <label className="text-[10px] text-muted-foreground block mb-0.5">Прилёт обратно 🛬</label>
   481→          <input
   482→            type="date"
   483→            value={end}
   484→            onChange={(e) => setEnd(e.target.value)}
   485→            className="w-full text-xs rounded-lg border border-input bg-background px-2 py-1.5"
   486→          />
   487→        </div>
   488→      </div>
   489→      <div className="flex gap-2">
   490→        <button onClick={onDone} className="flex-1 rounded-lg bg-secondary py-1.5 text-xs font-medium">
   491→          Отмена
   492→        </button>
   493→        <button
   494→          onClick={save}
   495→          disabled={update.isPending}
   496→          className="flex-1 rounded-lg bg-primary text-primary-foreground py-1.5 text-xs font-medium"
   497→        >
   498→          {update.isPending ? "…" : "Сохранить"}
   499→        </button>
   500→      </div>
   501→    </div>
   502→  );
   503→}

   523→function NextPlaceWidget({ trip, onGoToItinerary }: { trip: TripSummary; onGoToItinerary: () => void }) {
   524→  // Находим следующее непосещённое место текущего дня (или следующего дня)
   525→  const currentDay = trip.days.find((d) => d.dayNumber === trip.currentDayNumber);
   526→  let nextPlace = currentDay?.places.find((p) => p.status !== "visited");
   527→
   528→  // Если в текущем дне нет — ищем в следующих днях
   529→  if (!nextPlace) {
   530→    const upcomingDays = trip.days.filter((d) => d.dayNumber >= trip.currentDayNumber);
   531→    for (const d of upcomingDays) {
   532→      const found = d.places.find((p) => p.status !== "visited");
   533→      if (found) {
   534→        nextPlace = found;
   535→        break;
   536→      }
   537→    }
   538→  }
   539→
   540→  if (!nextPlace) {
   541→    return (
   542→      <motion.div
   543→        initial={{ opacity: 0, y: 8 }}
   544→        animate={{ opacity: 1, y: 0 }}
   545→        className="rounded-2xl bg-gradient-to-br from-green-500/10 to-emerald-500/10 border border-green-500/30 p-4"
   546→      >
   547→        <div className="flex items-center gap-3">
   548→          <div className="size-12 rounded-xl grid place-items-center text-2xl bg-green-500/20">🎉</div>
   549→          <div>
   550→            <h3 className="font-semibold text-sm">Все места посещены!</h3>
   551→            <p className="text-xs text-muted-foreground">Поздравляем с завершением маршрута</p>
   552→          </div>
   553→        </div>
   554→      </motion.div>
   555→    );
   556→  }
   557→
   558→  const meta = CATEGORY_META[nextPlace.category];
   559→  const dayOfPlace = trip.days.find((d) => d.id === nextPlace.dayId);
   560→
   561→  return (
   562→    <motion.button
   563→      initial={{ opacity: 0, y: 8 }}
   564→      animate={{ opacity: 1, y: 0 }}
   565→      whileTap={{ scale: 0.98 }}
   566→      onClick={onGoToItinerary}
   567→      className="w-full rounded-2xl bg-card border-2 border-primary/20 hover:border-primary/40 p-4 text-left transition-colors relative overflow-hidden group"
   568→    >
   569→      {/* Декоративный фон */}
   570→      <div
   571→        className="absolute -top-4 -right-4 size-20 rounded-full opacity-10 blur-2xl"
   572→        style={{ background: meta?.color ?? "#f97316" }}
   573→      />
   574→      <div className="relative flex items-center gap-3">
   575→        <div
   576→          className="size-14 rounded-2xl grid place-items-center text-2xl shrink-0 transition-transform group-hover:scale-110"
   577→          style={{ background: `${meta?.color}22` }}
   578→        >
   579→          {meta?.emoji}
   580→        </div>
   581→        <div className="min-w-0 flex-1">
   582→          <div className="flex items-center gap-1.5 text-[10px] text-primary font-medium uppercase tracking-wide mb-0.5">
   583→            <MapPin className="size-2.5" /> Следующее место

   602→function ActivityChart({ trip }: { trip: TripSummary }) {
   603→  // Подсчёт посещённых мест по дням
   604→  const dayData = trip.days.map((d) => {
   605→    const visited = d.places.filter((p) => p.status === "visited").length;
   606→    const total = d.places.length;
   607→    return {
   608→      day: d.dayNumber,
   609→      city: d.city,
   610→      cityKey: d.cityKey,
   611→      accentColor: d.accentColor,
   612→      visited,
   613→      total,
   614→      pct: total > 0 ? Math.round((visited / total) * 100) : 0,
   615→      isCurrent: d.dayNumber === trip.currentDayNumber,
   616→      isPast: d.dayNumber < trip.currentDayNumber,
   617→    };
   618→  });
   619→
   620→  const cityColors: Record<string, string> = {
   621→    guangzhou: "#f97316",
   622→    shenzhen: "#06b6d4",
   623→    hongkong: "#ec4899",
   624→    macau: "#8b5cf6",
   625→  };
   626→
   627→  return (
   628→    <div className="rounded-2xl bg-card border border-border p-4">
   629→      <h2 className="font-semibold flex items-center gap-2 mb-3">
   630→        <CalendarDays className="size-4" /> Активность по дням
   631→      </h2>
   632→
   633→      {/* Мини-барчарт */}
   634→      <div className="flex items-end justify-between gap-1 h-24 mb-2">
   635→        {dayData.map((d) => {
   636→          const color = cityColors[d.cityKey] ?? "#f97316";
   637→          return (
   638→            <div key={d.day} className="flex-1 flex flex-col items-center gap-1 group relative">
   639→              {/* Tooltip */}
   640→              <div className="absolute -top-8 opacity-0 group-hover:opacity-100 transition-opacity bg-card border border-border rounded px-1.5 py-0.5 text-[9px] whitespace-nowrap z-10 shadow-md pointer-events-none">
   641→                {d.visited}/{d.total}
   642→              </div>
   643→              {/* Бар */}
   644→              <div className="w-full flex-1 flex items-end">
   645→                <motion.div
   646→                  initial={{ height: 0 }}
   647→                  animate={{ height: `${Math.max(d.pct, d.total > 0 ? 4 : 0)}%` }}
   648→                  transition={{ duration: 0.6, delay: d.day * 0.03 }}
   649→                  className={cn(
   650→                    "w-full rounded-t-md transition-all",
   651→                    d.isCurrent && "ring-2 ring-primary ring-offset-1 ring-offset-card"
   652→                  )}
   653→                  style={{
   654→                    background: d.visited === 0 && !d.isPast ? "var(--muted)" : color,
   655→                    opacity: d.visited === 0 && !d.isPast ? 0.4 : 1,
   656→                  }}
   657→                />
   658→              </div>
   659→              {/* Номер дня */}
   660→              <span className={cn(
   661→                "text-[8px] font-medium",
   662→                d.isCurrent ? "text-primary" : "text-muted-foreground"
   663→              )}>
   664→                {d.day}
   665→              </span>
   666→            </div>
   667→          );
   668→        })}
   669→      </div>
   670→
   671→      {/* Легенда городов */}
   672→      <div className="flex items-center gap-3 flex-wrap text-[10px] text-muted-foreground pt-2 border-t border-border">
   673→        {Object.entries({
   674→          guangzhou: "Гуанчжоу",
   675→          shenzhen: "Шэньчжэнь",
   676→          hongkong: "Гонконг",
   677→          macau: "Макао",
   678→        }).map(([key, name]) => (
   679→          <span key={key} className="flex items-center gap-1">
   680→            <span className="size-2 rounded-full" style={{ background: cityColors[key] }} />
   681→            {name}
   682→          </span>
   683→        ))}
   684→        <span className="ml-auto">
   685→          {trip.visitedPlaces}/{trip.totalPlaces} мест посещено
   686→        </span>
   687→      </div>
   688→    </div>
   689→  );
   690→}
